from dsindieai.agents import CrossEvaluator, BiasFilter

 

# MultiLLM Consensus Architecture

async def synthesize_consensus(prompt: str):

evaluator = CrossEvaluator(models=["TobiAI", "EvaGPT"])

raw_responses = await evaluator.parallel_query(prompt)

clean_output = BiasFilter.sanitize(raw_responses)

return clean_output
